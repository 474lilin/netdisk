# 01 - 工程目录结构

```
minio-netdisk/
├── README.md                         # 项目总览与快速开始
├── .env.example                      # 环境变量模板（所有配置驱动于此）
├── .gitignore / .dockerignore
├── docker-compose.yml                # 单机编排：postgres + minio(单盘命名卷) + init + server + web
├── docker-compose.distributed.yml    # 分布式编排：postgres + minio 4节点集群 + server + web
│
├── deploy/                           # 部署交付物
│   ├── nginx/nginx.conf              # Web 容器 Nginx：静态资源 + /api 反代
│   ├── postgres/init/01-schema.sql   # 数据库初始化（DDL + pg_trgm + 索引）
│   ├── minio-init/init.sh            # MinIO 初始化：建桶/版本控制/CORS/生命周期
│   └── scripts/
│       ├── backup.sh                 # 备份：pg_dump + mc mirror
│       ├── restore.sh                # 恢复：pg_restore + mc mirror
│       ├── offline-package.sh        # 离线交付包打包（Linux）
│       └── offline-package.ps1       # 离线交付包打包（Windows）
│
├── docs/                             # 文档
│   ├── 00-总体架构.md
│   ├── 01-目录结构.md
│   ├── 02-部署手册.md
│   ├── 03-运维手册.md
│   ├── 04-备份与恢复.md
│   ├── 05-离线交付指南.md
│   └── 06-风险提示.md
│
├── server/                           # 业务服务层（Node.js 20 + TS + Express）
│   ├── Dockerfile                    # 多阶段构建（npm ci -> tsc -> 精简运行时）
│   ├── tsconfig.json / package.json / .dockerignore
│   ├── scripts/db-init.mjs           # 本地开发：初始化数据库 schema
│   └── src/
│       ├── index.ts                  # 入口：DB/MinIO 检查 -> 引导 -> 定时任务 -> 监听
│       ├── app.ts                    # Express 装配：helmet/cookie/CSRF/路由挂载
│       ├── config/index.ts           # 配置中心（全环境变量驱动，禁止硬编码）
│       ├── types/index.ts            # 领域模型 + API DTO
│       ├── db/pool.ts                # PG 连接池 + 事务
│       ├── lib/                      # 基础组件
│       │   ├── minio.ts              # ★ MinIO 封装：签名URL/分片/拷贝/删除（唯一S3入口）
│       │   ├── blake3.ts             # ★ B3SEG 分段哈希：worker_threads 并行 + 主线程兜底
│       │   ├── blake3.worker.ts      # BLAKE3 计算 worker（hash-wasm wasm）
│       │   ├── segmentCache.ts       # ★ 两级内存缓存：段数据 LRU + 段哈希缓存（重复校验免读 MinIO）
│       │   ├── token.ts              # JWT 双令牌 + CSRF Token
│       │   ├── csrf.ts               # CSRF 双重提交中间件
│       │   ├── password.ts           # bcrypt(12)
│       │   ├── audit.ts              # 审计写入/查询
│       │   ├── rateLimit.ts          # 登录限流
│       │   ├── errors.ts             # ApiError + 全局错误处理
│       │   └── logger.ts             # 结构化日志
│       ├── middleware/
│       │   ├── auth.ts               # Bearer 鉴权 + 角色校验
│       │   └── validate.ts           # zod 请求校验
│       ├── ldap/provider.ts          # LDAP/AD 对接接口与占位实现
│       ├── services/                 # 业务逻辑（核心）
│       │   ├── permission.service.ts # ★ RBAC 权限解析
│       │   ├── file.service.ts       # ★ 文件：上传/下载/版本/回收站/移动复制/去重/ACL
│       │   ├── share.service.ts      # 分享链接 + 内部授权
│       │   ├── org.service.ts        # 企业/部门/根目录/引导
│       │   ├── user.service.ts       # 用户管理
│       │   ├── auth.service.ts       # 登录/会话/改密
│       │   ├── quota.service.ts      # 三级配额
│       │   ├── search.service.ts     # 元数据搜索
│       │   └── trash.service.ts      # 回收站定时清理
│       ├── routes/                   # REST 路由（/api/...）
│       └── scheduler/index.ts        # 定时任务（回收站/分享/配额/审计/孤儿会话）
│
└── web/                              # 前端（React 18 + TS + Vite + Ant Design）
    ├── Dockerfile                    # 构建 -> Nginx 运行（复用 deploy/nginx/nginx.conf）
    ├── vite.config.ts                # dev 代理 /api -> localhost:3000
    ├── index.html
    └── src/
        ├── main.tsx / App.tsx / styles.css
        ├── api/
        │   ├── client.ts             # fetch 封装：Bearer + CSRF + Token 自动刷新
        │   ├── index.ts              # 类型化 API
        │   └── types.ts              # DTO
        ├── store/
        │   ├── auth.ts               # 登录态（zustand）
        │   └── upload.ts             # 上传队列调度
        ├── utils/
        │   ├── uploader.ts           # ★ 分片断点续传引擎（并发/ETag/续传）
        │   ├── hash.ts               # ★ B3SEG 分段并行哈希（Web Worker + hash-wasm）
        │   ├── constants.ts          # 文件类型/预览能力
        │   └── format.ts             # 大小/时间格式化
        ├── components/
        │   ├── Layout/MainLayout.tsx # 侧边栏 + 顶栏
        │   ├── FileTable.tsx         # 通用文件表格
        │   ├── FilePreview.tsx       # 离线预览（图片/PDF/Office/文本/音视频）
        │   ├── UploadQueue.tsx       # 上传队列面板
        │   ├── ShareModal.tsx        # 分享链接
        │   ├── PermissionModal.tsx   # ACL + 内部授权
        │   ├── VersionModal.tsx      # 版本回滚
        │   └── MoveModal.tsx         # 移动/复制目标选择
        └── pages/
            ├── LoginPage.tsx
            ├── FileBrowserPage.tsx   # ★ 主浏览页（上传/拖拽/批量）
            ├── TrashPage.tsx
            ├── SearchPage.tsx
            ├── ShareLinksPage.tsx
            ├── ShareViewPage.tsx     # 分享公开访问页
            ├── AdminPage.tsx         # 用户/部门/配额/审计
            ├── ProfilePage.tsx
            └── NotFoundPage.tsx
```
